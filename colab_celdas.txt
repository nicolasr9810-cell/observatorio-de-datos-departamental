CELDA 5 - CONFIGURAR NGROK

En el panel lateral de Colab abre Secrets (icono de llave) y crea:
  Nombre: NGROK_AUTHTOKEN
  Valor: tu token privado de ngrok
No guardes el token en una celda, un archivo ni en GitHub.

CELDA 6 - INICIAR STREAMLIT Y ABRIR EL TÚNEL

import subprocess, time
from pathlib import Path
from google.colab import userdata
from pyngrok import ngrok

token = userdata.get("NGROK_AUTHTOKEN")
if not token:
    raise RuntimeError("Falta NGROK_AUTHTOKEN en Colab Secrets.")
ngrok.set_auth_token(token)
log_path = Path("/content/streamlit.log")
log_file = log_path.open("w")
process = subprocess.Popen(
    ["streamlit", "run", "/content/observatorio/app.py", "--server.port", "8501",
     "--server.address", "0.0.0.0", "--server.headless", "true"],
    cwd="/content/observatorio", stdout=log_file, stderr=subprocess.STDOUT,
)
time.sleep(10)
if process.poll() is not None:
    log_file.flush()
    raise RuntimeError("Streamlit se detuvo: " + log_path.read_text(errors="replace")[-5000:])
public_url = ngrok.connect(8501, "http").public_url
print("URL temporal del Observatorio:", public_url)

CELDA 7 - CERRAR

ngrok.disconnect(public_url)
process.terminate()
log_file.close()
